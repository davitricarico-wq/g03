@startuml FL01_Cadastro_Completo_Familia_Moradia

title FL01 - Cadastro completo de familia, moradia e ocupacao

actor "Agente de Campo (A01)" as Agente
participant "Frontend PWA Mobile" as Frontend
participant "Cache Local\n(IndexedDB)" as Cache
participant "CadastroController" as Controller
participant "CadastroService" as Service
participant "CadastroRepository" as Repository
database "Banco de Dados" as DB

== Preenchimento em campo ==
Agente -> Frontend : Inicia novo cadastro
Frontend -> Agente : Exibe secoes de moradia,\nlocalizacao, familia, responsavel,\nmoradores, grupos prioritarios,\ngestante, pets e fotos

Agente -> Frontend : Preenche dados estruturais da moradia
Frontend -> Frontend : Captura GPS do dispositivo (RN04)

alt GPS permitido e disponivel
    Frontend -> Frontend : Preenche latitude e longitude
else GPS indisponivel ou negado
    Frontend -> Agente : Solicita CEP, logradouro,\nbairro, numero, cidade,\nUF e ponto de referencia
    Agente -> Frontend : Informa localizacao manual
end

Agente -> Frontend : Anexa ate 2 fotos\n(fachada e entorno)
Frontend -> Frontend : Valida RN04:\nfoto nao pode conter pessoas
Agente -> Frontend : Informa responsavel,\ndemais cidadaos, vulnerabilidades,\ngestacao quando houver e pets

== Envio ou persistencia local ==
Agente -> Frontend : Confirma cadastro

alt Dispositivo online
    Frontend -> Controller : POST /cadastros-completos\n{localizacao, moradia, familia,\nresponsavel, cidadaos, grupos,\ngestantes, pets, fotos}
    Controller -> Service : Validar payload, RN01 e RN04
    Service -> Service : Validar integridade:\nfamilia ativa deve ter responsavel\ne ocupacao ativa
    Service -> Repository : Abrir transacao

    Repository -> DB : INSERT localizacao
    DB --> Repository : id_localizacao
    Repository -> DB : INSERT moradia\n{id_localizacao, status='Ativa'}
    DB --> Repository : id_moradia
    Repository -> DB : INSERT familia\n{status_ativo=true}
    DB --> Repository : id_familia
    Repository -> DB : INSERT historico_ocupacao\n{id_familia, id_moradia,\ndata_entrada=hoje,\ndata_saida=NULL, status='Regular'}
    DB --> Repository : id_historico_ocupacao

    Repository -> DB : INSERT cidadao do responsavel\n{id_familia, status_cadastro=true}
    DB --> Repository : id_cidadao_responsavel
    Repository -> DB : INSERT responsavel\n{id_cidadao_responsavel, cpf,\nrenda, contato, NIS}
    DB --> Repository : id_responsavel

    loop Para cada morador dependente
        Repository -> DB : INSERT cidadao\n{id_familia, status_cadastro=true}
        DB --> Repository : id_cidadao

        opt Morador pertence a grupos prioritarios
            Repository -> DB : INSERT cidadao_grupo_prioritario\n{id_cidadao, id_grupo_prioritario}
            DB --> Repository : OK
        end

        opt Morador gestante
            Repository -> DB : INSERT gestante\n{id_cidadao, data_prevista,\ndata_inicio}
            DB --> Repository : OK
        end
    end

    loop Para cada pet informado
        Repository -> DB : INSERT pet\n{id_familia, tipo_pet,\nporte_pet, nome, observacoes}
        DB --> Repository : OK
    end

    loop Para cada foto de moradia
        Repository -> DB : INSERT foto_moradia\n{id_moradia, tipo_foto, url}
        DB --> Repository : OK
    end

    Repository --> Service : Commit da transacao
    Service --> Controller : HTTP 201 Created\n{id_familia, id_moradia,\nid_responsavel}
    Controller --> Frontend : HTTP 201 Created
    Frontend -> Agente : Exibe confirmacao do cadastro

else Dispositivo offline
    Frontend -> Cache : Salva cadastro completo\ncom UUID local e estado PENDENTE
    Cache --> Frontend : Persistido localmente
    Frontend -> Agente : Exibe "Salvo localmente,\naguardando sincronizacao"

    Cache -> Frontend : Ao reconectar, aciona sync
    Frontend -> Controller : POST /cadastros-completos\n{uuid_local, payload}
    note over Frontend, Service
      A sincronizacao deve ser idempotente.
      O UUID local evita duplicidade quando
      o envio for repetido por falha de rede.
    end note
end

== Falhas principais ==
alt CPF, email ou NIS duplicado
    Service --> Controller : HTTP 409 Conflict
    Controller --> Frontend : HTTP 409 Conflict
    Frontend -> Agente : Exibe opcao de buscar cadastro\nou revisar dados
else Campos obrigatorios invalidos
    Service --> Controller : HTTP 422 Unprocessable Entity
    Controller --> Frontend : Lista de campos invalidos
    Frontend -> Agente : Destaca campos para correcao
end

@startuml FL11_Regra_Transversal_Risco_Critico

title FL11 - Regra transversal de Risco Critico (RN05)

participant "Fluxo Chamador\n(FL02/FL03/FL04/FL05/FL07/FL09)" as Chamador
participant "Service" as Service
participant "Repository" as Repository
database "Banco de Dados" as DB
participant "Frontend" as Frontend
actor "Usuario" as Usuario

note over Chamador, Service
  RN05 nao e um fluxo independente.
  E aplicada quando a consulta integrada,
  mapa, filtro, relatorio ou atualizacao
  carrega moradia com ocupacao ativa.
end note

Chamador -> Service : Solicita avaliacao RN05\n{id_moradia}
Service -> Repository : Buscar moradia, ocorrencias\ne moradores ativos da ocupacao atual
Repository -> DB : SELECT indicador de historico_ocorrencia,\ncidadao.id_cidadao,\ngrupo_prioritario.nome\nFROM moradia\nJOIN historico_ocupacao\nJOIN familia\nJOIN cidadao\nLEFT JOIN cidadao_grupo_prioritario\nLEFT JOIN grupo_prioritario\nWHERE moradia.id_moradia=:id\nAND historico_ocupacao.data_saida IS NULL\nAND cidadao.status_cadastro=true
DB --> Repository : Dados para avaliacao
Repository --> Service : Dados para avaliacao

Service -> Service : Verificar condicao:\nhistorico_ocorrencia=true\nE algum morador em grupo\n'Mobilidade reduzida' ou 'Acamado'

alt Condicao RN05 satisfeita
    Service --> Chamador : risco_critico=true
    Chamador --> Frontend : Objeto com flag
    Frontend -> Usuario : Exibe "Risco Critico"\nno cabecalho/card
else Condicao RN05 nao satisfeita
    Service --> Chamador : risco_critico=false
    Chamador --> Frontend : Objeto sem flag critica
    Frontend -> Usuario : Exibe ficha normalmente
end

@enduml


@startuml FL12_Integridade_Familia_Responsavel_Ocupacao

title FL12 - Validacao transversal de integridade cadastral

participant "Fluxo Chamador\n(Cadastro, Atualizacao,\nArquivamento, Realocacao)" as Chamador
participant "Service" as Service
participant "Repository" as Repository
database "Banco de Dados" as DB
participant "Frontend" as Frontend
actor "Usuario" as Usuario

note over Service
  Regras derivadas das US13 e US14:
  familia ativa deve possuir responsavel ativo
  e ocupacao ativa em uma moradia valida.
end note

Chamador -> Service : Solicita validacao de integridade\n{id_familia}
Service -> Repository : Verificar familia ativa
Repository -> DB : SELECT familia\nWHERE id_familia=:id
DB --> Repository : Familia
Repository --> Service : Familia

alt Familia inativa
    Service --> Chamador : Integridade aprovada\npara contexto historico
else Familia ativa
    Service -> Repository : Verificar responsavel ativo
    Repository -> DB : SELECT responsavel, cidadao\nWHERE cidadao.id_familia=:id\nAND cidadao.status_cadastro=true\nAND responsavel.id_cidadao=cidadao.id_cidadao
    DB --> Repository : Responsavel ativo ou vazio
    Repository --> Service : Resultado

    Service -> Repository : Verificar ocupacao ativa
    Repository -> DB : SELECT historico_ocupacao, moradia\nWHERE id_familia=:id\nAND data_saida IS NULL\nAND moradia.status='Ativa'
    DB --> Repository : Ocupacao ativa ou vazio
    Repository --> Service : Resultado

    alt Sem responsavel ativo
        Service --> Chamador : HTTP 409 Conflict\n{erro='familia_sem_responsavel'}
        Chamador --> Frontend : Bloqueio de operacao
        Frontend -> Usuario : Solicita definir novo responsavel
    else Sem moradia ativa vinculada
        Service --> Chamador : HTTP 409 Conflict\n{erro='familia_sem_moradia_ativa'}
        Chamador --> Frontend : Bloqueio de operacao
        Frontend -> Usuario : Solicita vincular nova moradia\nou inativar familia
    else Integridade preservada
        Service --> Chamador : Validacao OK
    end
end

@enduml
